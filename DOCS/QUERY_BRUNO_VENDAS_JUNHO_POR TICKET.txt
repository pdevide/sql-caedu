SELECT TOP 1000
CODIGO_FILIAL,
FILIAIS.FILIAL,
DATA_VENDA,
CODIGO_BARRA,
LOJA_VENDA_PRODUTO.PRODUTO,
PRODUTOS.DESC_PRODUTO,
COR_PRODUTO,
QTDE,
PRECO_LIQUIDO,
CUSTO
FROM LOJA_VENDA_PRODUTO 
INNER JOIN FILIAIS ON FILIAIS.COD_FILIAL = LOJA_VENDA_PRODUTO.CODIGO_FILIAL
INNER JOIN PRODUTOS ON PRODUTOS.PRODUTO = LOJA_VENDA_PRODUTO.PRODUTO
where data_venda >= '20180601' and data_venda <= '20180630'
